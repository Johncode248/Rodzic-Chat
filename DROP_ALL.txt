-- ==============================================================================
-- PEŁNE WYCZYSZCZENIE BAZY (Postgres) -- USUWA WSZYSTKIE TABELE I DANE. NIEODWRACALNE.
--
-- Usuwa każdą tabelę ze schematu "public" (CASCADE: razem z indeksami, kluczami obcymi i zależnymi obiektami), więc
-- nie trzeba utrzymywać listy tabel i nie ma znaczenia kolejność. Można uruchamiać wielokrotnie, także na pustej bazie.
-- Schematy systemowe i inne schematy nie są ruszane. Zaktualizowane 2026-10-12.
--
-- Uruchomienie:   psql "<connection string>" -f DROP_ALL.txt
-- Potem:          odkomentuj //BigMigrate() w main.go, uruchom backend raz (tworzy tabele, klucze obce i ograniczenia
--                 -- patrz big.migrate.go), po starcie zakomentuj z powrotem.
-- ==============================================================================

BEGIN;

DO $$
DECLARE
  t record;
BEGIN
  FOR t IN SELECT tablename FROM pg_tables WHERE schemaname = 'public' LOOP
    EXECUTE format('DROP TABLE IF EXISTS %I.%I CASCADE', 'public', t.tablename);
  END LOOP;
END $$;

COMMIT;
